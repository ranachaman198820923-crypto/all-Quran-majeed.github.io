// آن لائن حدیث سرچ کرنے کا فنکشن async function searchHadithOnline() { const query = document.getElementById('hadithInput').value.trim(); const resultsContainer = document.getElementById('hadithResults'); if (!query) { resultsContainer.innerHTML = '

براہِ مہربانی کوئی لفظ یا موضوع لکھیں۔

'; return; } resultsContainer.innerHTML = '

تلاش جاری ہے...

'; try { // آن لائن حدیث API کے ذریعے سرچ const response = await fetch(`https://hadithapi.com/api/hadiths?apiKey=$2y$10$YourApiKeyHere&search=${encodeURIComponent(query)}`); const data = await response.json(); if (data.hadiths && data.hadiths.data.length > 0) { resultsContainer.innerHTML = ''; data.hadiths.data.forEach(item => { resultsContainer.innerHTML += `

${item.hadithArabic || ''}

${item.hadithUrdu || item.hadithEnglish || ''}

حوالہ: ${item.book.bookName} - حدیث نمبر: ${item.hadithNumber}
`; }); } else { // اگر API میں نہ ملے تو گوگل پر ڈائریکٹ تلاش کا لنک دیں resultsContainer.innerHTML = `

معذرت، ایپ کے آن لائن لوکل ڈیٹا بیس میں یہ لفظ نہیں ملا۔

گوگل پر "${query}" حدیث سرچ کرنے کے لیے یہاں کلک کریں
`; } } catch (error) { // انٹرنیٹ نہ ہونے یا ناکامی پر گوگل سرچ کا بیک اپ resultsContainer.innerHTML = `

آن لائن سرچ سے رابطہ نہیں ہو سکا۔

گوگل (Google) پر براہِ راست تلاش کریں
`; } }
